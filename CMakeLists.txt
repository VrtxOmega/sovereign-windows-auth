cmake_minimum_required(VERSION 3.24)
project(SovereignWindowsAuth VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This prototype requires MSVC and Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FIDO_ROOT "${CMAKE_SOURCE_DIR}/artifacts/libfido2/libfido2-1.17.0-win" CACHE PATH "Verified libfido2 SDK")
set(FIDO_BIN "${FIDO_ROOT}/Win64/Release/v143/dynamic")
add_library(swa_core STATIC src/core.cpp src/profile.cpp src/windows_auth.cpp src/current_user.cpp src/windows_pin.cpp)
target_include_directories(swa_core PUBLIC src "${FIDO_ROOT}/include")
target_compile_definitions(swa_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX SECURITY_WIN32)
target_compile_options(swa_core PUBLIC /W4 /WX /sdl /guard:cf /EHsc)
target_link_libraries(swa_core PUBLIC "${FIDO_BIN}/fido2.lib" bcrypt crypt32 advapi32 shell32 ole32 credui secur32 netapi32 shlwapi propsys)
add_executable(swa_probe src/probe.cpp)
target_link_libraries(swa_probe PRIVATE swa_core)
add_executable(swa_enroll src/enroll.cpp src/hello_probe.rc)
add_executable(swa_hello_probe src/hello_probe.cpp src/hello_probe.rc)
target_link_libraries(swa_hello_probe PRIVATE swa_core)
target_link_options(swa_hello_probe PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
add_executable(swa_signin_inspect src/signin_inspect.cpp)
target_link_libraries(swa_signin_inspect PRIVATE swa_core netapi32 ncrypt shlwapi propsys)
target_link_options(swa_signin_inspect PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
target_link_libraries(swa_enroll PRIVATE swa_core credui netapi32)
target_link_options(swa_enroll PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
add_library(SovereignCredentialProvider SHARED src/provider.cpp src/provider.def)
target_link_libraries(SovereignCredentialProvider PRIVATE swa_core credui secur32 shlwapi delayimp gdi32)
target_link_options(SovereignCredentialProvider PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /CETCOMPAT /DELAYLOAD:fido2.dll)
add_executable(swa_provider_host src/provider_host.cpp)
target_link_libraries(swa_provider_host PRIVATE swa_core credui shlwapi gdi32)
target_link_options(swa_provider_host PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
target_link_options(swa_probe PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
foreach(dll fido2.dll crypto-56.dll cbor.dll zlib1.dll)
  add_custom_command(TARGET swa_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FIDO_BIN}/${dll}" "$<TARGET_FILE_DIR:swa_probe>")
endforeach()
enable_testing()
add_executable(swa_windows_auth_test src/windows_auth_test.cpp)
target_link_libraries(swa_windows_auth_test PRIVATE swa_core)
target_link_options(swa_windows_auth_test PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
add_test(NAME windows_online_identity_format COMMAND swa_windows_auth_test)
add_test(NAME crypto_and_parser COMMAND swa_probe --self-test)
add_test(NAME credential_provider_contract COMMAND swa_provider_host "$<TARGET_FILE:SovereignCredentialProvider>")

# The filter experiment is an executable only: no registration or live DLL path.
add_executable(swa_filter_lab src/filter_lab.cpp)
target_compile_definitions(swa_filter_lab PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(swa_filter_lab PRIVATE /W4 /WX /sdl /guard:cf /EHsc)
target_link_libraries(swa_filter_lab PRIVATE ole32 advapi32)
target_link_options(swa_filter_lab PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
add_test(NAME credential_filter_lab_contract COMMAND swa_filter_lab --self-test)
